------------------------------------------------------------
-- ОПРЕДЕЛЕНИЕ ТЕКУЩЕЙ МИНИИГРЫ
-- Стратегии (по приоритету):
--   1) StringValue в ReplicatedStorage с именем CurrentMinigame/MinigameName/CurMinigame
--   2) StringValue / IntValue в ReplicatedStorage чьё .Value соответствует одной из миниигр
--   3) Сканирование PlayerGui — TextLabel текст которого совпадает с миниигрой
--   4) Сканирование Workspace на наличие модели с именем миниигры
------------------------------------------------------------
local function tryByReplicatedValue()
    -- Прямые имена
    for _,name in ipairs({"CurrentMinigame","MinigameName","CurMinigame","CurrentGame","Minigame","CurrentMap","Map","CurrentStatus","Status"}) do
        local v = ReplicatedStorage:FindFirstChild(name, true)
        if v and v:IsA("StringValue") and v.Value ~= "" then
            local val = tostring(v.Value)
            if Minigames[val] then return val end
            -- иногда в Status пишут "Now playing: X"
            for mg,_ in pairs(Minigames) do
                if string.find(val, mg, 1, true) then return mg end
            end
        end
    end
    -- Любой StringValue с подходящим значением
    for _,d in ipairs(ReplicatedStorage:GetDescendants()) do
        if d:IsA("StringValue") then
            local v = tostring(d.Value)
            if Minigames[v] then return v end
        end
    end
    return nil
end

local function tryByPlayerGui()
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return nil end
    for _,d in ipairs(pg:GetDescendants()) do
        if d:IsA("TextLabel") or d:IsA("TextButton") then
            local t = tostring(d.Text or "")
            if t ~= "" and #t < 60 then
                if Minigames[t] then return t end
                -- "Minigame: X" / "Now playing: X"
                local cap = t:match("[Mm]inigame[%s:]+(.+)$") or t:match("[Pp]laying[%s:]+(.+)$") or t:match("[Nn]ow[%s]*[Pp]laying[%s:]*(.+)$")
                if cap and Minigames[cap] then return cap end
            end
        end
    end
    return nil
end

local function tryByWorkspace()
    for mg,_ in pairs(Minigames) do
        if Workspace:FindFirstChild(mg, true) then return mg end
    end
    return nil
end

local function detectMinigame()
    return tryByReplicatedValue() or tryByPlayerGui() or tryByWorkspace() or "Lobby"
end

------------------------------------------------------------
-- FLY (новая версия, работает на ПК и телефоне)
-- ПК:  WASD + Space/Shift + камера
-- Тел: экранные кнопки ↑ ↓ ⬆ (вперёд по камере) + сам наводишь камерой
------------------------------------------------------------
local FlyLV, FlyAO, FlyAtt, FlyConn
local FlyGui, FlyBtnFwd, FlyBtnBack, FlyBtnUp, FlyBtnDown
local FlyMobileInput = {fwd = false, back = false, up = false, down = false}

local function destroyFlyGui()
    if FlyGui then FlyGui:Destroy(); FlyGui = nil end
    FlyBtnFwd, FlyBtnBack, FlyBtnUp, FlyBtnDown = nil, nil, nil, nil
end

local function buildFlyGui()
    destroyFlyGui()
    local pg = LocalPlayer:FindFirstChildOfClass("PlayerGui")
    if not pg then return end
    FlyGui = Instance.new("ScreenGui")
    FlyGui.Name = "EM_FlyControls"
    FlyGui.ResetOnSpawn = false
    FlyGui.IgnoreGuiInset = true
    FlyGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    FlyGui.Parent = pg

    local function mkBtn(name, text, pos)
        local b = Instance.new("TextButton")
        b.Name = name
        b.Size = UDim2.new(0, 64, 0, 64)
        b.Position = pos
        b.AnchorPoint = Vector2.new(0, 0)
        b.BackgroundColor3 = Color3.fromRGB(35, 35, 40)
        b.BackgroundTransparency = 0.25
        b.BorderSizePixel = 0
        b.Text = text
        b.TextColor3 = Color3.fromRGB(255,255,255)
        b.TextScaled = true
        b.Font = Enum.Font.GothamBold
        b.AutoButtonColor = false
        local c = Instance.new("UICorner", b); c.CornerRadius = UDim.new(0, 12)
        local s = Instance.new("UIStroke", b); s.Color = Color3.fromRGB(120,120,140); s.Thickness = 1.5; s.Transparency = 0.4
        b.Parent = FlyGui
        return b
    end

    -- Размещаем в правом нижнем углу
    FlyBtnUp   = mkBtn("Up",   "▲",   UDim2.new(1, -90,  1, -240))
    FlyBtnDown = mkBtn("Down", "▼",   UDim2.new(1, -90,  1, -90))
    FlyBtnFwd  = mkBtn("Fwd",  "GO",  UDim2.new(1, -170, 1, -165))
    FlyBtnBack = mkBtn("Back", "BACK",UDim2.new(1, -10,  1, -165))

    local function bind(btn, key)
        btn.InputBegan:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.Touch
                or i.UserInputType == Enum.UserInputType.MouseButton1 then
                FlyMobileInput[key] = true
                btn.BackgroundColor3 = Color3.fromRGB(80,140,255)
            end
        end)
        btn.InputEnded:Connect(function(i)
            if i.UserInputType == Enum.UserInputType.Touch
                or i.UserInputType == Enum.UserInputType.MouseButton1 then
                FlyMobileInput[key] = false
                btn.BackgroundColor3 = Color3.fromRGB(35, 35, 40)
            end
        end)
    end
    bind(FlyBtnFwd,  "fwd")
    bind(FlyBtnBack, "back")
    bind(FlyBtnUp,   "up")
    bind(FlyBtnDown, "down")
end

local function stopFly()
    State.Fly = false
    if FlyConn then FlyConn:Disconnect(); FlyConn = nil end
    if FlyLV then FlyLV:Destroy(); FlyLV = nil end
    if FlyAO then FlyAO:Destroy(); FlyAO = nil end
    if FlyAtt then FlyAtt:Destroy(); FlyAtt = nil end
    destroyFlyGui()
    for k,_ in pairs(FlyMobileInput) do FlyMobileInput[k] = false end
end

local function startFly()
    stopFly()
    local hrp = getHRP(); local hum = getHum()
    if not hrp or not hum then return end
    State.Fly = true

    -- Современные движители (работают даже там, где BodyVelocity убрали)
    FlyAtt = Instance.new("Attachment")
    FlyAtt.Name = "EM_FlyAtt"
    FlyAtt.Parent = hrp

    FlyLV = Instance.new("LinearVelocity")
    FlyLV.Name = "EM_FlyLV"
    FlyLV.Attachment0 = FlyAtt
    FlyLV.MaxForce = math.huge
    FlyLV.PlaneVelocity = Vector2.zero
    FlyLV.VectorVelocity = Vector3.zero
    FlyLV.ForceLimitMode = Enum.ForceLimitMode.PerAxis
    FlyLV.MaxAxesForce = Vector3.new(1e6, 1e6, 1e6)
    FlyLV.RelativeTo = Enum.ActuatorRelativeTo.World
    FlyLV.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
    FlyLV.Parent = hrp

    FlyAO = Instance.new("AlignOrientation")
    FlyAO.Name = "EM_FlyAO"
    FlyAO.Attachment0 = FlyAtt
    FlyAO.Mode = Enum.OrientationAlignmentMode.OneAttachment
    FlyAO.RigidityEnabled = true
    FlyAO.Parent = hrp

    -- Снимаем гравитацию через PlatformStand чтобы Humanoid не дёргал
    if hum then pcall(function() hum.PlatformStand = false end) end

    if IS_MOBILE then buildFlyGui() end

    FlyConn = RunService.RenderStepped:Connect(function()
        if not State.Fly then return end
        local cam = Workspace.CurrentCamera
        if not cam then return end
        local move = Vector3.zero

        -- Клавиатура
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + cam.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - cam.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + cam.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space)     then move = move + Vector3.new(0,1,0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move = move - Vector3.new(0,1,0) end

        -- Экранные кнопки (мобила/планшет)
        if FlyMobileInput.fwd  then move = move + cam.CFrame.LookVector end
        if FlyMobileInput.back then move = move - cam.CFrame.LookVector end
        if FlyMobileInput.up   then move = move + Vector3.new(0,1,0) end
        if FlyMobileInput.down then move = move - Vector3.new(0,1,0) end

        if move.Magnitude > 0 then move = move.Unit end

        if FlyLV then FlyLV.VectorVelocity = move * State.FlySpeed end
        if FlyAO then FlyAO.CFrame = cam.CFrame end
    end)
end

------------------------------------------------------------
-- HOVER — зависнуть в воздухе и не упасть
-- Поднимает на HoverHeight studs и удерживает там
-- (используется для миниигр где надо избегать предметов/падений)
------------------------------------------------------------
local HoverConn, HoverAnchor, HoverCFrame
local function stopHover()
    State.Hover = false
    if HoverConn then HoverConn:Disconnect(); HoverConn = nil end
    HoverAnchor = nil
    HoverCFrame = nil
    local hum = getHum()
    if hum then
        pcall(function() hum.PlatformStand = false end)
    end
    local hrp = getHRP()
    if hrp then
        pcall(function() hrp.Anchored = false end)
    end
    -- удалить старые ограничители, если остались с прошлых версий
    local char = LocalPlayer.Character
    if char then
        for _, n in ipairs({"EM_HoverAtt","EM_HoverAP","EM_HoverAO"}) do
            local x = char:FindFirstChild(n, true)
            if x then x:Destroy() end
        end
    end
end

local function startHover(heightOverride)
    stopHover()
    local hrp = getHRP(); local hum = getHum()
    if not hrp or not hum then return end
    State.Hover = true
    local height = heightOverride or State.HoverHeight or 30

    -- Поднимаем персонажа вверх и запоминаем точку
    local lifted = hrp.CFrame + Vector3.new(0, height, 0)
    hrp.CFrame = lifted
    HoverAnchor = lifted.Position
    HoverCFrame = lifted

    -- Гасим Humanoid, чтобы он не тянул вниз / не дёргался
    pcall(function() hum.PlatformStand = true end)

    -- Жёстко удерживаем CFrame каждый кадр — это работает на 100%
    HoverConn = RunService.Stepped:Connect(function()
        if not State.Hover then return end
        local p = getHRP(); local h = getHum()
        if not p or not h then return end
        -- если включён Fly — не мешаем ему
        if State.Fly then return end
        -- Движок отшвырнул далеко от точки зависания => телепорт конца раунда.
        if HoverAnchor and (p.Position - HoverAnchor).Magnitude > 130 then
            State.RoundEndSignal = true
            stopHover()
            return
        end
        pcall(function()
            p.AssemblyLinearVelocity  = Vector3.zero
            p.AssemblyAngularVelocity = Vector3.zero
            -- сохраняем поворот персонажа, фиксируем только позицию
            local rot = p.CFrame - p.CFrame.Position
            p.CFrame = CFrame.new(HoverAnchor) * rot
            h.PlatformStand = true
        end)
    end)
end

-- ===== Kill Aura (универсальный) =====
local KillAuraFx = {}  -- Player -> {part, size, trans, col, mass}

local function clearKillAuraFx()
    for plr, fx in pairs(KillAuraFx) do
        if fx and fx.part and fx.part.Parent then
            pcall(function()
                fx.part.Size = fx.size
                fx.part.Transparency = fx.trans
                fx.part.CanCollide = fx.col
                fx.part.Massless = fx.mass
            end)
        end
    end
    KillAuraFx = {}
end

local function applyAuraHitbox(plr, size)
    if not plr or not plr.Character then return end
    local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    if KillAuraFx[plr] and KillAuraFx[plr].part == hrp then return end
    KillAuraFx[plr] = {
        part  = hrp,
        size  = hrp.Size,
        trans = hrp.Transparency,
        col   = hrp.CanCollide,
        mass  = hrp.Massless,
    }
    pcall(function()
        hrp.Size = Vector3.new(size, size, size)
        hrp.Transparency = 1
        hrp.CanCollide = false
        hrp.Massless = true
    end)
end

local function startKillAura(opts)
    stopCustom()
    CustomActive[opts.flag] = true
    setGodMode(true)
    State.WalkSpeed = 28; applySpeed()

    task.spawn(function()
        local isActive = function() return CustomActive[opts.flag] end
        if not waitForMinigameStart(isActive) then return end
        if not isActive() then return end

        if opts.waitSeconds and opts.waitSeconds > 0 then
            pcall(function()
                Rayfield:Notify({
                    Title = opts.label or "Kill Aura",
                    Content = ("Жду %d секунд..."):format(opts.waitSeconds),
                    Duration = 3,
                })
            end)
            local n = math.floor(opts.waitSeconds * 10)
            for _ = 1, n do
                if not isActive() then return end
                task.wait(0.1)
            end
        end
        if not isActive() then return end

        local myTeam = LocalPlayer.Team
        local hitbox = opts.hitboxSize or 30
        local range  = opts.range or 80

        pcall(function()
            Rayfield:Notify({
                Title = opts.label or "Kill Aura",
                Content = opts.teamBased
                    and ("Команда: "..(myTeam and myTeam.Name or "?")..". Аура по врагам.")
                    or "FFA — аура по всем.",
                Duration = 4,
            })
        end)

        while isActive() do
            local sword = equipAnyTool()
            local foundClose = false
            local me = getHRP()
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer then
                    local skip = false
                    if opts.teamBased and myTeam and plr.Team == myTeam then
                        skip = true
                    end
                    if not skip then
                        local c   = plr.Character
                        local hum = c and c:FindFirstChildOfClass("Humanoid")
                        local hrp = c and c:FindFirstChild("HumanoidRootPart")
                        if c and hum and hum.Health > 0 and hrp then
                            applyAuraHitbox(plr, hitbox)
                            if me and (hrp.Position - me.Position).Magnitude < range then
                                foundClose = true
                            end
                        end
                    end
                end
            end
            if sword and foundClose then
                pcall(function() sword:Activate() end)
            end
            task.wait(0.12)
        end
    end)
end

-- эвристика: похожа ли часть на кольцо (для Ring Diver)
local function isLikelyRing(p)
    if not p:IsA("BasePart") then return false end
    if p.Transparency >= 1 then return false end
    local n = p.Name:lower()
    if n:find("ring") or n:find("hoop") or n:find("torus") then return true end
    local mesh = p:FindFirstChildOfClass("SpecialMesh")
    if mesh then
        local id = tostring(mesh.MeshId or ""):lower()
        if id:find("ring") or id:find("torus") or id:find("hoop") then return true end
    end
    return false
end

-- эвристика: ярко-зелёная большая площадка (финиш в Turntable Scurry)
local function isLikelyGreenFinish(p)
    if not p:IsA("BasePart") then return false end
    if p.Transparency >= 1 then return false end
    local c = p.Color
    -- ярко-зелёный: G доминирует
    if not (c.G > 0.55 and c.R < 0.6 and c.B < 0.6 and (c.G - c.R) > 0.2 and (c.G - c.B) > 0.2) then
        return false
    end
    -- размер платформы — не мелкая деталь
    if p.Size.X < 4 or p.Size.Z < 4 then return false end
    return true
end

-- Nudge and Budge: мэш левой/правой кнопки очень быстро
local function findNudgeButtons()
    local cam = Workspace.CurrentCamera
    local vp = (cam and cam.ViewportSize) or Vector2.new(1920, 1080)
    local cands = {}
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then return nil, nil end
    for _, gui in ipairs(pg:GetDescendants()) do
        if (gui:IsA("ImageButton") or gui:IsA("TextButton")) and gui.Visible and gui.Active ~= false then
            local visible = true
            local p = gui.Parent
            while p and p ~= pg do
                if p:IsA("GuiObject") and not p.Visible then visible = false; break end
                p = p.Parent
            end
            if visible then
                local pos  = gui.AbsolutePosition
                local size = gui.AbsoluteSize
                local cx, cy = pos.X + size.X/2, pos.Y + size.Y/2
                if cy > vp.Y * 0.45 and size.X > 30 and size.Y > 30 then
                    table.insert(cands, {btn = gui, x = cx, y = cy, area = size.X * size.Y})
                end
            end
        end
    end
    if #cands < 2 then return nil, nil end
    table.sort(cands, function(a,b) return a.area > b.area end)
    local top = {}
    for i = 1, math.min(8, #cands) do table.insert(top, cands[i]) end
    table.sort(top, function(a,b) return a.x < b.x end)
    return top[1].btn, top[#top].btn
end

local function fireButton(btn)
    if not btn then return end
    pcall(function() if firesignal then firesignal(btn.MouseButton1Click) end end)
    pcall(function() if firesignal then firesignal(btn.MouseButton1Down)  end end)
    pcall(function() if firesignal then firesignal(btn.MouseButton1Up)    end end)
    pcall(function() if firesignal then firesignal(btn.Activated)         end end)
    pcall(function()
        if firesignal then
            firesignal(btn.InputBegan, {
                UserInputType = Enum.UserInputType.MouseButton1,
                UserInputState = Enum.UserInputState.Begin,
            })
            firesignal(btn.InputEnded, {
                UserInputType = Enum.UserInputType.MouseButton1,
                UserInputState = Enum.UserInputState.End,
            })
        end
    end)
end

-- Ждём пока игрок реально телепортируется на карту миниигры
-- (HRP сместится больше чем на 50 studs от стартовой позиции)
local function waitForMinigameStart(isActive, timeout)
    -- Считаем, что раунд начался, когда игрока ТЕЛЕПОРТНУЛО на карту.
    -- Детектим двумя способами, чтобы не зависнуть до таймаута:
    --   1) текущая позиция далеко от запомненного лобби (State.LobbyPos) —
    --      срабатывает даже если applyStrategy вызвали УЖЕ после телепорта;
    --   2) либо заметное смещение от точки, где стартовала стратегия.
    local hrp = getHRP()
    local startPos = hrp and hrp.Position or nil
    local t0 = tick()
    timeout = timeout or 12
    while (tick() - t0) < timeout do
        if isActive and not isActive() then return false end
        local p = getHRP()
        if p then
            -- уже далеко от лобби — точно на карте
            if State.LobbyPos and (p.Position - State.LobbyPos).Magnitude > 60 then
                task.wait(0.3)
                return true
            end
            -- заметный телепорт от точки запуска стратегии
            if startPos and (p.Position - startPos).Magnitude > 50 then
                task.wait(0.3)
                return true
            end
        end
        task.wait(0.1)
    end
    return true -- по таймауту тоже срабатываем (на всякий случай)
end

local function startBlockHunt()
    stopCustom()
    CustomActive.blockhunt = true
    setGodMode(true)
    State.WalkSpeed = 50; applySpeed()
    task.spawn(function()
        local isActive = function() return CustomActive.blockhunt end
        if not waitForMinigameStart(isActive) then return end
        if not isActive() then return end

        local myTeam = LocalPlayer.Team
        local myColor = myTeam and myTeam.TeamColor and myTeam.TeamColor.Color
        pcall(function()
            Rayfield:Notify({
                Title = "Block Hunt",
                Content = myTeam
                    and ("Команда: " .. myTeam.Name .. ". Жду 15с (placement), потом ТП по кубам врага.")
                    or "Команда не определена — буду ломать все цветные кубы.",
                Duration = 5,
            })
        end)

        for _ = 1, 150 do
            if not isActive() then return end
            task.wait(0.1)
        end
        if not isActive() then return end

        local destroyed = setmetatable({}, {__mode = "k"})
        local function colorDist(a, b)
            return math.abs(a.R - b.R) + math.abs(a.G - b.G) + math.abs(a.B - b.B)
        end
        local function saturation(c)
            return math.max(c.R, c.G, c.B) - math.min(c.R, c.G, c.B)
        end
        local function isCube(p)
            if not p:IsA("BasePart") or not p.Anchored then return false end
            local s = p.Size
            if s.X < 1 or s.X > 6 then return false end
            if math.abs(s.X - s.Y) > 0.6 then return false end
            if math.abs(s.Y - s.Z) > 0.6 then return false end
            return true
        end
        local function isEnemyColor(c)
            if myColor then
                return colorDist(c, myColor) > 0.6
            else
                return saturation(c) > 0.35
            end
        end

        local function touchBlock(target)
            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            if not hrp then return end
            pcall(function()
                hrp.AssemblyLinearVelocity = Vector3.zero
                hrp.AssemblyAngularVelocity = Vector3.zero
                hrp.CFrame = CFrame.new(target.Position)
            end)
            task.wait(0.05)
            if firetouchinterest then
                local parts = { hrp }
                for _, name in ipairs({ "LeftFoot", "RightFoot", "Torso", "UpperTorso", "LowerTorso", "HumanoidRootPart" }) do
                    local p = char:FindFirstChild(name)
                    if p and p ~= hrp then table.insert(parts, p) end
                end
                for _, p in ipairs(parts) do
                    pcall(firetouchinterest, p, target, 0)
                    pcall(firetouchinterest, p, target, 1)
                end
            end
        end

        while isActive() do
            local me = getHRP()
            if not me then
                task.wait(0.3)
            else
                local cands = {}
                for _, d in ipairs(Workspace:GetDescendants()) do
                    if not isActive() then break end
                    if not destroyed[d] and isCube(d) then
                        local sat = saturation(d.Color)
                        if sat >= 0.35 and isEnemyColor(d.Color) then
                            table.insert(cands, d)
                        end
                    end
                end
                if #cands > 0 then
                    table.sort(cands, function(a, b)
                        return (a.Position - me.Position).Magnitude < (b.Position - me.Position).Magnitude
                    end)
                    local target = cands[1]
                    touchBlock(target)
                    destroyed[target] = true
                    task.wait(0.18)
                else
                    task.wait(0.35)
                end
            end
        end
    end)
end

        else

                for _, plr in ipairs(Players:GetPlayers()) do

                end
                if myTeamCount == 1 then amJuggernaut = true end

                Title = "Juggernaut",
                Content = amJuggernaut and "Ты Джаггернаут — ТП к каждому и убиваем." or "Ты выживший — авто-аим по Джаггернауту.",

local function startCakeDelivery()
    stopCustom()
    CustomActive.cake = true
    setGodMode(true)
    State.WalkSpeed = 60; applySpeed()
    task.spawn(function()
        local isActive = function() return CustomActive.cake end
        if not waitForMinigameStart(isActive) then return end
        if not isActive() then return end
        task.wait(1.5)

        -- читаем номер из PlayerGui
        local function readMyNumber()
            local pg = LocalPlayer:FindFirstChild("PlayerGui")
            if not pg then return nil end
            local best, bestScore = nil, -1
            for _, gui in ipairs(pg:GetDescendants()) do
                if gui:IsA("TextLabel") and gui.Visible then
                    -- проверяем что родитель видим
                    local p = gui.Parent
                    local vis = true
                    while p and p ~= pg do
                        if p:IsA("GuiObject") and not p.Visible then vis = false; break end
                        p = p.Parent
                    end
                    if vis then
                        local txt = (gui.Text or ""):match("^%s*(%-?%d+)%s*$")
                        if txt then
                            local n = tonumber(txt)
                            if n and n >= 0 and n <= 999 then
                                local size = gui.AbsoluteSize
                                local score = size.X * size.Y
                                if score > bestScore then
                                    bestScore = score
                                    best = n
                                end
                            end
                        end
                    end
                end
            end
            return best
        end

        local function findDoor(num)
            -- ищем дверь с этим номером в Workspace
            local numStr = tostring(num)
            local candidates = {}
            for _, d in ipairs(Workspace:GetDescendants()) do
                if d:IsA("BasePart") then
                    local n = d.Name
                    -- имя точно равно числу или содержит "_N", "N", "Door N"
                    if n == numStr
                       or n:lower():find("door[ _]?" .. numStr .. "$")
                       or n:lower():find("door[ _]?" .. numStr .. "[^%d]")
                       or n:find("[^%d]" .. numStr .. "$")
                       or n:find("^" .. numStr .. "$")
                    then
                        table.insert(candidates, d)
                    end
                end
                -- SurfaceGui/BillboardGui с TextLabel = num на двери
                if d:IsA("TextLabel") and (d.Text or ""):match("^%s*"..numStr.."%s*$") then
                    local part = d:FindFirstAncestorOfClass("BasePart")
                          or (d.Parent and d.Parent:FindFirstAncestorOfClass("BasePart"))
                    if part then table.insert(candidates, part) end
                end
            end
            if #candidates == 0 then return nil end
            -- берём самый большой (двери обычно большие)
            table.sort(candidates, function(a,b)
                return (a.Size.X * a.Size.Y * a.Size.Z) > (b.Size.X * b.Size.Y * b.Size.Z)
            end)
            return candidates[1]
        end

        local lastNum = nil
        while isActive() do
            local num = readMyNumber()
            local me = getHRP()
            if num and me then
                if num ~= lastNum then
                    lastNum = num
                    pcall(function()
                        Rayfield:Notify({Title="Cake Delivery", Content="Номер: "..num.." → ищу дверь.", Duration=3})
                    end)
                end
                local door = findDoor(num)
                if door then
                    pcall(function()
                        me.AssemblyLinearVelocity = Vector3.zero
                        me.CFrame = door.CFrame * CFrame.new(0, 0, -3)
                    end)
                end
            end
            task.wait(0.3)
        end
    end)
end

        local done = setmetatable({}, {__mode = "k"})
        local BOMB_WORDS = {"bomb","tnt","explos","danger","trap","fuse","boom","mine","dynamite","barrel","skull","c4","death","kill"}

        local function nameHasBomb(obj)
            if not obj then return false end
            local n = tostring(obj.Name):lower()
            for _, w in ipairs(BOMB_WORDS) do if n:find(w, 1, true) then return true end end
            return false
        end

        local function fireNode(node)
            if done[node] then return end
            done[node] = true
            if node:IsA("ClickDetector") then
                pcall(function() if fireclickdetector then fireclickdetector(node, 0) end end)
                pcall(function() if fireclickdetector then fireclickdetector(node, 1) end end)
                pcall(function() if firesignal then firesignal(node.MouseClick, LocalPlayer) end end)
            elseif node:IsA("ProximityPrompt") then
                pcall(function() if fireproximityprompt then fireproximityprompt(node) end end)
            end
            local me = getHRP()
            local part = node.Parent
            if me and part and part:IsA("BasePart") then
                pcall(function() if firetouchinterest then firetouchinterest(me, part, 0); firetouchinterest(me, part, 1) end end)
            end
        end

        while CustomActive.mining do
            for _, d in ipairs(Workspace:GetDescendants()) do
                if not CustomActive.mining then break end
                if (d:IsA("ClickDetector") or d:IsA("ProximityPrompt")) and not done[d] then
                    if nameHasBomb(d) or nameHasBomb(d.Parent) then
                        done[d] = true   -- бомбу не трогаем никогда
                    else
                        fireNode(d)
                    end
                end
            end

        end
    end)

-- ===== Solo Swordsman =====
local function findEquippedTool(char)
    if not char then return nil end
    for _, t in ipairs(char:GetChildren()) do
        if t:IsA("Tool") then return t end
    end
    return nil
end

local function findToolInBackpack()
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if not bp then return nil end
    for _, t in ipairs(bp:GetChildren()) do
        if t:IsA("Tool") then return t end
    end
    return nil
end

local function equipAnyTool()
    local char = LocalPlayer.Character
    local equipped = findEquippedTool(char)
    if equipped then return equipped end
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil end
    local tool = findToolInBackpack()
    if tool then
        pcall(function() hum:EquipTool(tool) end)
        task.wait(0.1)
        return findEquippedTool(LocalPlayer.Character)
    end
    return nil
end

local function getAliveEnemies()
    local res = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local c = plr.Character
            local h = c and c:FindFirstChildOfClass("Humanoid")
            local hrp = c and c:FindFirstChild("HumanoidRootPart")
            if c and h and h.Health > 0 and hrp then
                table.insert(res, {plr = plr, char = c, hum = h, hrp = hrp})
            end
        end
    end
    return res
end

local function startSoloSwordsman()
    stopCustom()
    CustomActive.swordsman = true
    setGodMode(true)
    State.WalkSpeed = 30; applySpeed()

    task.spawn(function()
        if not waitForMinigameStart(function() return CustomActive.swordsman end) then return end
        if not CustomActive.swordsman then return end

        pcall(function()
            Rayfield:Notify({Title = "Solo Swordsman", Content = "ТП на карту. Жду 12 секунд...", Duration = 3})
        end)
        for _ = 1, 120 do
            if not CustomActive.swordsman then return end
            task.wait(0.1)
        end
        if not CustomActive.swordsman then return end

        pcall(function()
            Rayfield:Notify({Title = "Solo Swordsman", Content = "Готов. Сканирую меч / врагов.", Duration = 3})
        end)

        while CustomActive.swordsman do
            local sword = equipAnyTool()
            if not sword then
                -- меча нет — висим в воздухе
                if not State.Hover then startHover(40) end
                task.wait(0.5)
            else
                if State.Hover then stopHover() end
                local enemies = getAliveEnemies()
                if #enemies == 0 then
                    if not State.Hover then startHover(20) end
                    task.wait(0.4)
                else
                    local myHRP = getHRP()
                    if myHRP then
                        for _, e in ipairs(enemies) do
                            applyAuraHitbox(e.plr, 25)
                        end
                        table.sort(enemies, function(a, b)
                            return (a.hrp.Position - myHRP.Position).Magnitude
                                 < (b.hrp.Position - myHRP.Position).Magnitude
                            end)
                        local target = enemies[1]
                        local attackStart = tick()
                        while CustomActive.swordsman
                              and target.hum
                              and target.hum.Parent
                              and target.hum.Health > 0
                              and target.char
                              and target.char.Parent
                              and (tick() - attackStart) < 6 do
                            local me = getHRP()
                            if not me then break end
                            local behind = target.hrp.CFrame * CFrame.new(0, 0, 3)
                            pcall(function()
                                me.AssemblyLinearVelocity = Vector3.zero
                                me.CFrame = behind
                            end)
                            pcall(function() sword:Activate() end)
                            task.wait(0.12)
                        end
                    end
                end
            end
        end
    end)
end

-- ===== Above Water =====
-- Платформа врага держится на опорах/башнях. Целимся ПОД вражескую команду
-- (туда, где опоры) и непрерывно стреляем имеющимся оружием — платформа обвалится,
-- враги падают в воду. Себя держим в GodMode + AntiFling, чтобы не утонуть.
local function startAboveWater()
    stopCustom()
    CustomActive.abovewater = true
    setGodMode(true)
    setAntiFling(true)
    task.spawn(function()
        local isActive = function() return CustomActive.abovewater end
        if not waitForMinigameStart(isActive) then return end
        if not isActive() then return end
        pcall(function() Rayfield:Notify({
            Title   = "Above Water",
            Content = "Целюсь по опорам вражеской платформы и стреляю — она обвалится.",
            Duration = 5,
        }) end)

        local myTeam = LocalPlayer.Team

        -- точка прицеливания: центр вражеской команды, но чуть НИЖЕ —
        -- туда, где опоры/башни держат их платформу
        local function enemyAimPoint()
            local sum, n = Vector3.zero, 0
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr ~= LocalPlayer and (not myTeam or plr.Team ~= myTeam) then
                    local c = plr.Character
                    local hrp = c and c:FindFirstChild("HumanoidRootPart")
                    local hum = c and c:FindFirstChildOfClass("Humanoid")
                    if hrp and hum and hum.Health > 0 then
                        sum = sum + hrp.Position
                        n = n + 1
                    end
                end
            end
            if n == 0 then return nil end
            local center = sum / n
            return center - Vector3.new(0, 10, 0) -- метим в основание платформы
        end

        while isActive() do
            local me   = getHRP()
            local tool = equipAnyTool()
            local aim  = enemyAimPoint()
            if me and aim then
                local cam = Workspace.CurrentCamera
                pcall(function()
                    me.CFrame = CFrame.lookAt(me.Position, aim)
                    me.AssemblyAngularVelocity = Vector3.zero
                end)
                if cam then
                    pcall(function() cam.CFrame = CFrame.lookAt(cam.CFrame.Position, aim) end)
                end
                if tool then pcall(function() tool:Activate() end) end
            end
            task.wait(0.1)
        end
    end)
end

-- ===== Accurate Archery =====
-- Стрельба из лука по мишеням. Ищем ближайшую мишень (по имени), целимся камерой
-- и HRP точно в её центр и непрерывно жмём Activate (стреляем).
local function startAccurateArchery()
    stopCustom()
    CustomActive.archery = true
    task.spawn(function()
        local isActive = function() return CustomActive.archery end
        if not waitForMinigameStart(isActive) then return end
        if not isActive() then return end
        pcall(function() Rayfield:Notify({
            Title   = "Accurate Archery",
            Content = "Авто-прицел в центр мишени + автострельба из лука.",
            Duration = 5,
        }) end)

        local TARGET_WORDS = {"target","bullseye","bull","centre","center","circle","ring","board","dartboard","apple"}
        local function hasAny(name, words)
            for _, w in ipairs(words) do if name:find(w, 1, true) then return true end end
            return false
        end

        -- ищем «яблочко»: часть с подходящим именем, ближе и мельче = предпочтительнее
        local function findTargetPoint()
            local me = getHRP()
            if not me then return nil end
            local best, bestScore
            for _, dsc in ipairs(Workspace:GetDescendants()) do
                if dsc:IsA("BasePart") and dsc.Transparency < 1 then
                    local n = dsc.Name:lower()
                    if hasAny(n, TARGET_WORDS) then
                        local dist = (dsc.Position - me.Position).Magnitude
                        if dist > 3 and dist < 600 then
                            local score = dist + dsc.Size.Magnitude * 3
                            if not bestScore or score < bestScore then
                                bestScore, best = score, dsc
                            end
                        end
                    end
                end
            end
            return best and best.Position or nil
        end

        while isActive() do
            local me   = getHRP()
            local tool = equipAnyTool()
            local aim  = findTargetPoint()
            if me and aim then
                local cam = Workspace.CurrentCamera
                pcall(function() me.CFrame = CFrame.lookAt(me.Position, aim) end)
                if cam then
                    pcall(function() cam.CFrame = CFrame.lookAt(cam.CFrame.Position, aim) end)
                end
                if tool then
                    pcall(function() tool:Activate() end)
                end
            end
            task.wait(0.15)
        end
    end)
end

-- классификация цвета части в имя: red / green / blue / yellow
local function rgbName(c)
    if not c then return nil end
    local r, g, b = c.R, c.G, c.B
    if r > 0.5 and g > 0.45 and b < 0.45 and math.abs(r - g) < 0.45 then return "yellow" end
    if r > 0.45 and (r - g) > 0.18 and (r - b) > 0.18 then return "red" end
    if g > 0.45 and (g - r) > 0.18 and (g - b) > 0.18 then return "green" end
    if b > 0.45 and (b - r) > 0.18 and (b - g) > 0.18 then return "blue" end
    return nil
end

local function startCubeFactory()
    stopCustom()
    CustomActive.cubefactory = true
    setGodMode(true)
    task.spawn(function()
        if not waitForMinigameStart(function() return CustomActive.cubefactory end) then return end
        pcall(function() Rayfield:Notify({Title="Cube Factory", Content="Спамлю кнопку совпадения — ловлю каждое окно цвета.", Duration=4}) end)

        local function fireNode(node)
            if node:IsA("ClickDetector") then
                pcall(function() if fireclickdetector then fireclickdetector(node, 0) end end)
                pcall(function() if firesignal then firesignal(node.MouseClick, LocalPlayer) end end)
            elseif node:IsA("ProximityPrompt") then
                pcall(function() if fireproximityprompt then fireproximityprompt(node) end end)
            elseif node:IsA("GuiButton") then
                pcall(function() if firesignal then firesignal(node.MouseButton1Click) end end)
                pcall(function() if firesignal then firesignal(node.Activated) end end)
            end
        end

        while CustomActive.cubefactory do
            local me = getHRP()
            for _, d in ipairs(Workspace:GetDescendants()) do
                if not CustomActive.cubefactory then break end
                if d:IsA("ClickDetector") or d:IsA("ProximityPrompt") then
                    local part = d.Parent
                    if part and part:IsA("BasePart") and me then
                        if (part.Position - me.Position).Magnitude < 70 then fireNode(d) end
                    else
                        fireNode(d)
                    end
                end
            end
            local pg = LocalPlayer:FindFirstChild("PlayerGui")
            if pg then
                for _, g in ipairs(pg:GetDescendants()) do
                    if not CustomActive.cubefactory then break end
                    if g:IsA("GuiButton") and g.Visible then
                        local n = (tostring(g.Name) .. " " .. tostring(g.Text or "")):lower()
                        if n:find("cube") or n:find("match") or n:find("stamp") or n:find("press") or n:find("factory") or n:find("go") then
                            fireNode(g)
                        end
                    end
                end
            end
            task.wait(0.05)
        end
    end)
end

-- Rampant Rhythms: автотап по цвету бита, когда он доходит до кнопки
local function startRampantRhythms()
    stopCustom()
    CustomActive.rhythm = true
    task.spawn(function()
        if not waitForMinigameStart(function() return CustomActive.rhythm end) then return end
        pcall(function() Rayfield:Notify({Title="Rampant Rhythms", Content="Автотап по цвету бита у кнопки (red/green/blue/yellow).", Duration=4}) end)

        local function findButtons()
            local me = getHRP()
            local btns = {}
            for _, d in ipairs(Workspace:GetDescendants()) do
                if d:IsA("ClickDetector") then
                    local part = d.Parent
                    if part and part:IsA("BasePart") then
                        local cn = rgbName(part.Color)
                        if cn and (not me or (part.Position - me.Position).Magnitude < 90) then
                            local prev = btns[cn]
                            if not prev or part.Size.Magnitude > prev.size then
                                btns[cn] = {cd = d, part = part, pos = part.Position, size = part.Size.Magnitude}
                            end
                        end
                    end
                end
            end
            return btns
        end

        local function fireBtn(b)
            pcall(function() if fireclickdetector then fireclickdetector(b.cd, 0) end end)
            pcall(function() if firesignal then firesignal(b.cd.MouseClick, LocalPlayer) end end)
        end

        local cooldown = {}
        while CustomActive.rhythm do
            local btns = findButtons()
            for _, d in ipairs(Workspace:GetDescendants()) do
                if not CustomActive.rhythm then break end
                if d:IsA("BasePart") and d.Transparency < 1 then
                    local cn = rgbName(d.Color)
                    if cn and btns[cn] then
                        local b = btns[cn]
                        if d ~= b.part and d.Size.Magnitude < b.size then
                            local dist = (d.Position - b.pos).Magnitude
                            if dist < 7 then
                                local t = tick()
                                if not cooldown[cn] or (t - cooldown[cn]) > 0.12 then
                                    cooldown[cn] = t
                                    fireBtn(b)
                                end
                            end
                        end
                    end
                end
            end
            task.wait(0.03)
        end
    end)
end

local PendingStrategyId = 0

local function applyStrategy(strat, name)
    clearAnchor()
    stopCustom()
    State.BypassActive = true
    PendingStrategyId = PendingStrategyId + 1

    -- сначала проверяем индивидуальный обход
    if name and CustomBypass[name] then
        CustomBypass[name]()
        return
    end

    if strat == "obby" then
        State.WalkSpeed = 40
        State.JumpPower = 60
        applySpeed(); applyJump()
        setInfJump(true)
        if not State.Fly then startFly() end

    elseif strat == "survival" then
        State.WalkSpeed = 24
        applySpeed()
        setInfJump(true)
        setGodMode(true)
        local myId = PendingStrategyId
        task.spawn(function()
            local active = function() return State.BypassActive and PendingStrategyId == myId end
            if not waitForMinigameStart(active) then return end
            if not active() then return end
            startHover(35)
        end)

    elseif strat == "race" then
        State.WalkSpeed = 60
        State.JumpPower = 70
        applySpeed(); applyJump()
        setInfJump(true)

    elseif strat == "pvp" then
        State.WalkSpeed = 32
        State.JumpPower = 60
        applySpeed(); applyJump()
        setInfJump(true)
        setGodMode(true)

    elseif strat == "team" then
        State.WalkSpeed = 28
        applySpeed()
        setInfJump(true)

    elseif strat == "defense" then
        setGodMode(true)
        local myId = PendingStrategyId
        task.spawn(function()
            local active = function() return State.BypassActive and PendingStrategyId == myId end
            if not waitForMinigameStart(active) then return end
            if not active() then return end
            startHover(15)
        end)

    elseif strat == "collect" then
        State.WalkSpeed = 50
        applySpeed()
        setNoclip(true)
        setInfJump(true)
        if not State.Fly then startFly() end

    elseif strat == "puzzle" then
        setInfJump(true)
        setGodMode(true)
        local myId = PendingStrategyId
        task.spawn(function()
            local active = function() return State.BypassActive and PendingStrategyId == myId end
            if not waitForMinigameStart(active) then return end
            if not active() then return end
            startHover(40)
        end)
    end
end

local function clearStrategy()
    State.BypassActive = false
    clearAnchor()
    stopCustom()
    if State.Fly then stopFly() end
    if State.Hover then stopHover() end
    setNoclip(false)
    setInfJump(false)
    setGodMode(false)
    State.WalkSpeed = 16
    State.JumpPower = 50
    applySpeed(); applyJump()
    -- Сбросить и визуальное состояние тумблеров в UI, чтобы они показывали OFF
    for _, flag in ipairs({"Fly","Hover","Noclip","InfJump","GodMode"}) do
        pcall(function()
            local f = Rayfield and Rayfield.Flags and Rayfield.Flags[flag]
            if f and f.Set then f:Set(false) end
        end)
    end
end

-- ЦИКЛ ОПРЕДЕЛЕНИЯ МИНИИГРЫ + AUTO-BYPASS
------------------------------------------------------------
task.spawn(function()
    while task.wait(0.4) do
        local mg = detectMinigame()
        State.CurrentMinigame = mg
        local d = Minigames[mg]

        -- === КОНЕЦ РАУНДА ПО ТЕЛЕПОРТУ ===
        -- lock/hover отпустили нас, потому что движок телепортнул на спавн.
        -- Работает даже когда персонаж был «заморожен» — мы поймали сам телепорт.
        if State.RoundEndSignal then
            State.RoundEndSignal = false
            State.RoundGen = State.RoundGen + 1
            clearStrategy()
            pcall(function() MGLabel:Set({Title = "Minigame: Lobby", Content = "Раунд закончился — обход снят (поймал телепорт на спавн)."}) end)
            pcall(function() Rayfield:Notify({Title = "Bypass снят", Content = "Конец раунда — отпустил, едешь на спавн.", Duration = 3}) end)
        end

        -- Пока в лобби — постоянно запоминаем позицию лобби (для детекта старта/конца).
        if mg == "Lobby" or not d then
            local hrp = getHRP()
            if hrp then State.LobbyPos = hrp.Position end
        end

        -- === ДЕТЕКТ КОНЦА РАУНДА ПО ПОЗИЦИИ ===
        -- Если обход активен, а игрока вернуло почти в точку лобби — раунд кончился.
        -- Ловит конец, даже если название миниигры ещё не успело смениться.
        if State.BypassActive and State.LobbyPos then
            local hrp = getHRP()
            if hrp and (hrp.Position - State.LobbyPos).Magnitude < 35 then
                State.RoundGen = State.RoundGen + 1
                clearStrategy()
                pcall(function() MGLabel:Set({Title = "Minigame: Lobby", Content = "Раунд закончился — обход снят (вернулись в лобби)."}) end)
                pcall(function() Rayfield:Notify({Title = "Bypass снят", Content = "Вернулись в лобби — обход отключён.", Duration = 3}) end)
            end
        end

        if mg ~= State.LastMinigame then
            local prev = State.LastMinigame
            State.LastMinigame = mg
            State.RoundGen = State.RoundGen + 1  -- отменяет отложенный запуск прошлого раунда
            local content
            if d then
                content = ("Категория: %s\nСтратегия: %s\n%s"):format(d.cat, d.strat, STRAT_LABEL[d.strat] or "")
            else
                content = "Сейчас в лобби или миниигра не распознана."
            end
            pcall(function()
                MGLabel:Set({Title = "Minigame: " .. mg, Content = content})
            end)

            -- Карта ПРОПАЛА (раунд закончился / вернулись в лобби) — снять весь обход.
            local leftMap = (prev ~= "" and prev ~= "Lobby" and (mg == "Lobby" or not d))
            if leftMap then
                clearStrategy()
                pcall(function()
                    Rayfield:Notify({
                        Title = "Bypass снят",
                        Content = "Карта пропала — обход «" .. prev .. "» отключён.",
                        Duration = 3,
                    })
                end)
            end

            -- Зашли на НОВУЮ карту: ждём StartDelay секунд (раунд успевает реально начаться),
            -- затем применяем обход. Ожидание отменяется, если миниигра сменилась,
            -- раунд закончился (RoundGen вырос) или Auto-Bypass выключили.
            if State.AutoBypass and d then
                local myGen  = State.RoundGen
                local thisMg = mg
                local delay  = State.StartDelay or 17
                pcall(function()
                    Rayfield:Notify({
                        Title = "Minigame: " .. mg,
                        Content = ("Жду %d сек до старта раунда…"):format(delay),
                        Duration = 4,
                    })
                end)
                task.spawn(function()
                    local t0 = tick()
                    while (tick() - t0) < delay do
                        task.wait(0.2)
                        if State.RoundGen ~= myGen then return end
                        if not State.AutoBypass then return end
                        if State.CurrentMinigame ~= thisMg then return end
                    end
                    if State.RoundGen == myGen and State.AutoBypass and State.CurrentMinigame == thisMg then
                        applyStrategy(d.strat, thisMg)
                    end
                end)
            end
        end
    end
end)
